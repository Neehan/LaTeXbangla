% Compare automatic Latin runs with direct selection of their companion font.
% A width mismatch catches wrong faces, size, ligatures or discretionary text.
\newcommand\checkfontpair[4]{%
  \setbox0=\hbox{#2\bengalifont #4}%
  \setbox2=\hbox{#2#3#4}%
  \ifdim\wd0=\wd2
    \typeout{FONT-PAIR-#1: equal}%
  \else
    \errmessage{Font pair #1 differs: \the\wd0\space versus \the\wd2}%
  \fi}

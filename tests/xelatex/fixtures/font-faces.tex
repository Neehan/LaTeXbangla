\newcommand\facesample{শিক্ষার্থী, দৃষ্টি, স্বচ্ছ, স্বাধীনতা, পূর্ণসংখ্যা, ক্ষুদ্র, জ্ঞান।}
\newcommand\facerows{%
  Regular:\quad\facesample\par\smallskip
  Bold:\quad\textbf{\facesample}\par\smallskip
  Italic:\quad\textit{\facesample}\par\smallskip
  Bold italic:\quad\textbf{\textit{\facesample}}\par
}
\setlength{\parindent}{0pt}
\subsection*{Body: 11 pt}
{\fontsize{11}{16.5}\selectfont\facerows}
\subsection*{Footnotes: 8 pt}
{\fontsize{8}{12}\selectfont\facerows}
\subsection*{Headings: 16 pt}
{\fontsize{16}{24}\selectfont\facerows}
